Project Akademik_Plus {
  database_type: 'Model Logiczny Danych (Implementacja Fizyczna: PostgreSQL/MSSQL + NoSQL)'
  Note: 'Pełny, zaktualizowany schemat bazy danych modelujący architekturę i encje. Model ten logicznie jest ujednolicony, jak niżej. Fizycznie encje powiązane z wiadomościami i usterkami lądują w bazie dokumentowej w oparciu o obiektowe referencje, reszta struktury - w bazie MSSQL.'
}

Enum "user_role" {
  "student"
  "admin"
}

Enum "room_standard" {
  "basic"
  "standard"
  "premium"
}

Enum "room_status" {
  "available"
  "full"
  "maintenance"
}

Enum "payment_status" {
  "paid"
  "pending"
  "overdue"
}

Enum "issue_category" {
  "plumbing"
  "electrical"
  "furniture"
  "heating"
  "other"
}

Enum "issue_status" {
  "open"
  "in-progress"
  "resolved"
  "closed"
}

Enum "issue_priority" {
  "low"
  "medium"
  "high"
}

TableGroup "MSSQL_Rozliczenia_i_Pokoje" {
  users
  students
  rooms
  residence_history
  payments
}

TableGroup "MongoDB_Usterki_i_Czaty" {
  issues
  issue_messages
  conversations
  chat_messages
}

Table "users" {
  "id" varchar [pk, note: 'Unikalny identyfikator użytkownika']
  "name" varchar
  "email" varchar [unique, not null]
  "password_hash" varchar [note: 'Hash hasła do logowania']
  "role" user_role [not null, default: 'student']
}

Table "students" {
  "id" varchar [pk]
  "user_id" varchar [note: 'Relacja 1:1 - każdy student to użytkownik']
  "student_matriculation_id" varchar [unique, note: 'Numer albumu (np. STU001)']
  "phone_number" varchar
  "room_id" varchar [note: 'Obecnie przypisany pokój']
  "bed_number" int
  "check_in_date" date
  "check_out_date" date
}

Table "rooms" {
  "id" varchar [pk]
  "number" varchar [unique]
  "floor" int
  "capacity" int [note: 'Całkowita liczba miejsc']
  "occupied" int [default: 0, note: 'Aktualnie zajęte miejsca']
  "standard" room_standard
  "price_per_month" decimal(10,2)
  "equipment" json [note: 'Lista wyposażenia (np. łóżko, biurko) w formacie JSON']
  "status" room_status [default: 'available']
  "sole_use" boolean [default: false, note: 'Możliwość wynajmu na wyłączność']
}

Table "residence_history" {
  "id" varchar [pk]
  "student_id" varchar
  "room_id" varchar
  "check_in_date" date
  "check_out_date" date [note: 'Null oznacza aktywny pobyt']
}

Table "payments" {
  "id" varchar [pk]
  "student_id" varchar
  "amount" decimal(10,2)
  "due_date" date [note: 'Termin płatności']
  "paid_date" date [note: 'Rzeczywista data wpłaty']
  "status" payment_status [default: 'pending']
  "month" varchar [note: 'Opłacany miesiąc (np. Marzec)']
  "year" int
}

Table "issues" {
  "id" varchar [pk]
  "student_id" varchar
  "room_id" varchar
  "title" varchar
  "description" text
  "category" issue_category
  "status" issue_status [default: 'open']
  "priority" issue_priority [default: 'medium']
  "created_at" timestamp
  "resolved_at" timestamp
}

Table "issue_messages" {
  "id" varchar [pk]
  "issue_id" varchar
  "sender_id" varchar
  "sender_name" varchar [note: 'Zdenormalizowane dla celów optymalizacji widoku']
  "sender_role" user_role
  "message" text
  "timestamp" timestamp
}

Table "conversations" {
  "id" varchar [pk]
  "student_id" varchar [note: 'Relacja 1:1 (admin rozmawia bezpośrednio ze studentem)']
  "student_name" varchar [note: 'Zdenormalizowane']
  "last_message" text
  "last_message_time" timestamp
  "unread_count" int [default: 0]
}

Table "chat_messages" {
  "id" varchar [pk]
  "conversation_id" varchar
  "sender_id" varchar
  "sender_name" varchar
  "sender_role" user_role
  "message" text
  "timestamp" timestamp
  "read" boolean [default: false]
}

Ref:"users"."id" < "students"."user_id"
Ref:"rooms"."id" < "students"."room_id"
Ref:"students"."id" < "residence_history"."student_id"
Ref:"rooms"."id" < "residence_history"."room_id"
Ref:"students"."id" < "payments"."student_id"
Ref:"students"."id" < "issues"."student_id"
Ref:"rooms"."id" < "issues"."room_id"
Ref:"issues"."id" < "issue_messages"."issue_id"
Ref:"users"."id" < "issue_messages"."sender_id"
Ref:"students"."id" < "conversations"."student_id"
Ref:"conversations"."id" < "chat_messages"."conversation_id"
Ref:"users"."id" < "chat_messages"."sender_id"
